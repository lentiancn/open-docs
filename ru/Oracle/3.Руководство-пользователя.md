### Подключение SQL*Plus

```sql
-- Локальное подключение
sqlplus username/password@localhost:1521/SID

-- Удаленное подключение
sqlplus username/password@hostname:1521/service_name

-- Подключение системного администратора
sqlplus / as sysdba
```

## Базовые операции SQL

### Язык определения данных (DDL)
```sql
-- Создание таблицы
CREATE TABLE employees (
    employee_id NUMBER PRIMARY KEY,
    first_name VARCHAR2(50),
    last_name VARCHAR2(50) NOT NULL,
    email VARCHAR2(100) UNIQUE,
    hire_date DATE DEFAULT SYSDATE,
    salary NUMBER(8,2)
);

-- Изменение структуры таблицы
ALTER TABLE employees ADD department_id NUMBER;
ALTER TABLE employees MODIFY salary NUMBER(10,2);
ALTER TABLE employees DROP COLUMN department_id;

-- Удаление таблицы
DROP TABLE employees;
```

### Язык манипулирования данными (DML)
```sql
-- Вставка данных
INSERT INTO employees (employee_id, first_name, last_name, email, salary)
VALUES (101, 'John', 'Doe', 'john.doe@example.com', 5000.00);

-- Запрос данных
SELECT employee_id, first_name, last_name, salary
FROM employees
WHERE salary > 4000
ORDER BY last_name;

-- Обновление данных
UPDATE employees
SET salary = salary * 1.1
WHERE department_id = 10;

-- Удаление данных
DELETE FROM employees
WHERE employee_id = 101;
```

## Программирование PL/SQL

### Хранимые процедуры
```sql
CREATE OR REPLACE PROCEDURE update_employee_salary(
    p_employee_id IN NUMBER,
    p_new_salary IN NUMBER
)
IS
BEGIN
    UPDATE employees
    SET salary = p_new_salary
    WHERE employee_id = p_employee_id;
    
    IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Employee not found');
    END IF;
    
    COMMIT;
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;
END;
/
```

### Функции
```sql
CREATE OR REPLACE FUNCTION get_employee_name(
    p_employee_id IN NUMBER
) RETURN VARCHAR2
IS
    v_full_name VARCHAR2(101);
BEGIN
    SELECT first_name || ' ' || last_name
    INTO v_full_name
    FROM employees
    WHERE employee_id = p_employee_id;
    
    RETURN v_full_name;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN NULL;
END;
/
```

### Триггеры
```sql
CREATE OR REPLACE TRIGGER employees_audit_trigger
    BEFORE INSERT OR UPDATE OR DELETE ON employees
    FOR EACH ROW
DECLARE
    v_action VARCHAR2(10);
BEGIN
    IF INSERTING THEN
        v_action := 'INSERT';
    ELSIF UPDATING THEN
        v_action := 'UPDATE';
    ELSE
        v_action := 'DELETE';
    END IF;
    
    INSERT INTO employees_audit (
        employee_id, action, action_date, user_name
    ) VALUES (
        NVL(:OLD.employee_id, :NEW.employee_id),
        v_action,
        SYSDATE,
        USER
    );
END;
/
```

## Оптимизация производительности

### Управление индексами
```sql
-- Создание B-дерево индекса
CREATE INDEX idx_employees_last_name ON employees(last_name);

-- Создание битовой карты индекса (подходит для столбцов с низкой кардинальностью)
CREATE BITMAP INDEX idx_employees_department ON employees(department_id);

-- Создание функционального индекса
CREATE INDEX idx_employees_upper_name ON employees(UPPER(last_name));
```

### Анализ плана выполнения
```sql
-- Просмотр плана выполнения

SELECT * FROM TABLE(DBMS_XPLAN.DISPLAY);
```

## Резервное копирование и восстановление

### Резервное копирование RMAN
```bash
# Подключение к RMAN
rman target /

# Полное резервное копирование базы данных
BACKUP DATABASE;

# Инкрементное резервное копирование
BACKUP INCREMENTAL LEVEL 1 DATABASE;

# Резервное копирование архивных журналов
BACKUP ARCHIVELOG ALL;
```

### Экспорт/Импорт Data Pump
```bash
# Экспорт всей базы данных
expdp system/password FULL=Y DIRECTORY=DATA_PUMP_DIR DUMPFILE=full_db.dmp

# Экспорт конкретной схемы
expdp hr/hr SCHEMAS=HR DIRECTORY=DATA_PUMP_DIR DUMPFILE=hr_schema.dmp

# Импорт данных
impdp system/password DIRECTORY=DATA_PUMP_DIR DUMPFILE=hr_schema.dmp
```

## Функции высокой доступности

### Real Application Clusters (RAC)
- Архитектура кластера с несколькими узлами
- Конфигурация общего хранилища
- Балансировка нагрузки сервисов
- Автоматическое переключение при отказе

### Data Guard
- Синхронизация основной и резервной баз данных
- Физическая и логическая резервная копия
- Автоматическое переключение при отказе
- Разделение чтения и записи

## Новые функции Oracle Database 26ai

Oracle Database 26ai вводит функции автоматизации на основе ИИ:
- **Автоматическая настройка SQL**: ИИ анализирует шаблоны запросов и автоматически оптимизирует
- **Прогнозирующее обслуживание**: предсказывает потенциальные проблемы и предоставляет ранние предупреждения
- **Интеллектуальное управление ресурсами**: динамически регулирует распределение памяти и процессора
- **Улучшенный мониторинг безопасности**: обнаружение аномалий на основе ИИ

Подробную документацию по новым функциям следует смотреть на официальном сайте Oracle.

## Лучшие практики

### Настройка безопасности
- Использовать надежные политики паролей
- Включить прозрачное шифрование данных (TDE)
- Настроить детальный контроль доступа
- Регулярно проводить аудит активности базы данных

### Оптимизация производительности
- Разумно проектировать структуры таблиц и индексы
- Использовать привязанные переменные для избежания жесткого анализа
- Регулярно собирать статистику
- Мониторить и оптимизировать медленные запросы

### Высокая доступность
- Реализовывать соответствующие стратегии резервного копирования
- Настроить Data Guard или RAC
- Регулярно тестировать процедуры восстановления
- Мониторить состояние работоспособности системы
