# SQLite 使用手册

## 命令行接口

### 启动 SQLite

```bash
sqlite3 [数据库文件名]
sqlite3 mydatabase.db
```

如果数据库文件不存在，SQLite 会自动创建。

### 退出 SQLite

```sql
.quit
.exit
```

### 常用命令

```sql
-- 显示帮助
.help

-- 列出所有表
.tables

-- 显示表结构
.schema [table-name]

-- 导出数据库
.output stdout
.dump

-- 导入 SQL 文件
.read filename.sql

-- 切换输出格式
.mode column
.mode line
.mode csv
.mode tabs

-- 执行单行命令
sqlite3 mydb.db "SELECT * FROM users;"
```

## SQL 语法

### 数据类型

SQLite 支持以下数据类型：

- `NULL`：空值
- `INTEGER`：有符号整数
- `REAL`：浮点数
- `TEXT`：文本字符串
- `BLOB`：二进制数据

### 创建数据库和表

```sql
-- 创建数据库
sqlite3 company.db

-- 创建表
CREATE TABLE employees (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    age INTEGER,
    salary REAL,
    department TEXT
);

-- 创建带约束的表
CREATE TABLE orders (
    order_id INTEGER PRIMARY KEY,
    customer_id INTEGER NOT NULL,
    order_date DATE DEFAULT (datetime('now')),
    total REAL CHECK (total >= 0),
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);
```

### 插入数据

```sql
-- 单行插入
INSERT INTO employees (name, age, salary, department) 
VALUES ('张三', 30, 5000.00, 'IT');

-- 多行插入
INSERT INTO employees (name, age, salary, department) 
VALUES 
    ('李四', 28, 4500.00, 'HR'),
    ('王五', 35, 6000.00, 'IT'),
    ('赵六', 32, 5500.00, 'Sales');

-- 使用默认值
INSERT INTO orders (customer_id) VALUES (1);
```

### 查询数据

```sql
-- 基本查询
SELECT * FROM employees;

-- 条件查询
SELECT name, salary FROM employees WHERE age > 30;

-- 排序
SELECT * FROM employees ORDER BY salary DESC;

-- 分组统计
SELECT department, COUNT(*), AVG(salary) 
FROM employees 
GROUP BY department;

-- 连接查询
SELECT e.name, d.department_name 
FROM employees e 
INNER JOIN departments d ON e.department_id = d.id;

-- 子查询
SELECT * FROM employees 
WHERE salary > (SELECT AVG(salary) FROM employees);
```

### 更新数据

```sql
-- 更新特定记录
UPDATE employees 
SET salary = 5500.00 
WHERE name = '张三';

-- 批量更新
UPDATE employees 
SET salary = salary * 1.1 
WHERE department = 'IT';
```

### 删除数据

```sql
-- 删除特定记录
DELETE FROM employees WHERE age < 25;

-- 删除所有记录（保留表结构）
DELETE FROM employees;

-- 完全清空表（更快）
TRUNCATE TABLE employees; -- SQLite 不支持，使用 DROP 和 CREATE
```

### 修改表结构

```sql
-- 添加列
ALTER TABLE employees ADD COLUMN email TEXT;

-- 重命名表
ALTER TABLE employees RENAME TO staff;

-- 注意：SQLite 的 ALTER TABLE 功能有限
-- 要删除列或更复杂的修改需要新建表并迁移数据
```

## 高级功能

### 视图

```sql
-- 创建视图
CREATE VIEW high_salary_employees AS
SELECT name, salary, department 
FROM employees 
WHERE salary > 5000;

-- 使用视图
SELECT * FROM high_salary_employees;

-- 删除视图
DROP VIEW high_salary_employees;
```

### 索引

```sql
-- 创建索引
CREATE INDEX idx_employees_name ON employees(name);
CREATE INDEX idx_employees_salary ON employees(salary);

-- 创建唯一索引
CREATE UNIQUE INDEX idx_employees_email ON employees(email);

-- 删除索引
DROP INDEX idx_employees_name;
```

### 事务

```sql
-- 开始事务
BEGIN TRANSACTION;

-- 执行操作
INSERT INTO employees (name, salary) VALUES ('新员工', 5000);
UPDATE employees SET salary = salary * 1.1 WHERE department = 'IT';

-- 提交事务
COMMIT;

-- 回滚事务
BEGIN TRANSACTION;
-- ... 操作 ...
ROLLBACK;
```

### 触发器

```sql
-- 创建触发器
CREATE TRIGGER update_salary_log
AFTER UPDATE ON employees
BEGIN
    INSERT INTO salary_log (employee_id, old_salary, new_salary, update_time)
    VALUES (OLD.id, OLD.salary, NEW.salary, datetime('now'));
END;

-- 删除触发器
DROP TRIGGER update_salary_log;
```

### 备份和恢复

```sql
-- 备份数据库
.dump > backup.sql

-- 恢复数据库
.read backup.sql

-- 使用命令行备份
sqlite3 mydb.db ".dump" | gzip > backup.sql.gz
```

## 函数参考

### 聚合函数

```sql
SELECT 
    COUNT(*) as total,           -- 计数
    SUM(salary) as total_salary, -- 求和
    AVG(salary) as avg_salary,   -- 平均值
    MAX(salary) as max_salary,   -- 最大值
    MIN(salary) as min_salary    -- 最小值
FROM employees;
```

### 字符串函数

```sql
SELECT 
    LENGTH(name) as name_length,     -- 字符串长度
    UPPER(name) as upper_name,       -- 转大写
    LOWER(name) as lower_name,       -- 转小写
    SUBSTR(name, 1, 3) as substring, -- 子字符串
    REPLACE(name, ' ', '_') as replaced, -- 替换
    TRIM(name) as trimmed            -- 去空白
FROM employees;
```

### 日期和时间函数

```sql
SELECT 
    date('now') as today,           -- 当前日期
    time('now') as now,             -- 当前时间
    datetime('now') as datetime,    -- 当前日期时间
    date('now', '+1 day') as tomorrow, -- 明天
    strftime('%Y-%m-%d', 'now') as formatted -- 格式化日期
```

### 条件函数

```sql
SELECT 
    name,
    CASE 
        WHEN salary > 5000 THEN 'High'
        WHEN salary > 3000 THEN 'Medium'
        ELSE 'Low'
    END as salary_level
FROM employees;

SELECT 
    name,
    COALESCE(email, 'no email') as contact
FROM employees;
```

## 配置选项

### PRAGMA 语句

```sql
-- 查看数据库信息
PRAGMA database_list;

-- 查看表信息
PRAGMA table_info(employees);

-- 设置页面大小
PRAGMA page_size = 4096;

-- 设置日志模式
PRAGMA journal_mode = WAL;

-- 启用外键约束
PRAGMA foreign_keys = ON;

-- 查看表索引
PRAGMA index_list(employees);
```

### 连接字符串参数

```bash
# 使用 URI 格式指定连接参数
sqlite3 "file:mydb.db?mode=ro&cache=shared"

# 参数说明：
# - mode=ro：只读模式
# - mode=rw：读写模式
# - cache=shared：共享缓存
# - cache=private：私有缓存
# - synchronous=OFF：非同步模式（更快但不安全）
# - synchronous=NORMAL：正常同步
# - synchronous=FULL：完全同步（最安全）
```

## 性能优化

### EXPLAIN 命令

```sql
-- 查看执行计划
EXPLAIN QUERY PLAN SELECT * FROM employees WHERE salary > 5000;

-- 显示详细执行步骤
EXPLAIN SELECT * FROM employees;
```

### 分析查询

```sql
-- 收集统计信息
ANALYZE;

-- 查看统计信息
SELECT * FROM sqlite_stat1;
```

### 适当使用索引

```sql
-- 为经常查询的列创建索引
CREATE INDEX idx_employees_dept_salary 
ON employees(department, salary);

-- 覆盖索引查询（只查询索引列）
CREATE INDEX idx_name ON employees(name);
SELECT name FROM employees WHERE name LIKE '张%';
```


## 错误处理

### 常见错误

```sql
-- 主键冲突
INSERT INTO employees (id, name) VALUES (1, '张三');
-- Error: PRIMARY KEY is not unique

-- 非空约束
INSERT INTO employees (age) VALUES (30);
-- Error: NOT NULL constraint failed: employees.name

-- 外键约束
INSERT INTO orders (customer_id) VALUES (999);
-- Error: FOREIGN KEY constraint failed
```

### 错误码

```bash
# SQLite 返回的错误码
sqlite3 mydb.db "SELECT * FROM nonexistent;" 2>&1
# 输出：Error: no such table: nonexistent
```
